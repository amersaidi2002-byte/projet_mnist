cmake_minimum_required(VERSION 3.16)
project(embedded_digits VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_VIDEO_APPS "Build image/video applications (requires OpenCV 4)" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(digit_cnn inference_c/CNN/neural_net_cnn.cpp)
add_library(digit_mlp inference_c/MLP/neural_network.cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/test_core.cpp)
  target_link_libraries(core_tests PRIVATE digit_cnn digit_mlp)
  add_test(NAME core COMMAND core_tests ${CMAKE_SOURCE_DIR}/inference_c/CNN)
  add_executable(cnn_reference tests/cnn_reference.cpp)
  target_link_libraries(cnn_reference PRIVATE digit_cnn)
  add_executable(mlp_reference tests/mlp_reference.cpp)
  target_link_libraries(mlp_reference PRIVATE digit_mlp)
endif()
if(BUILD_VIDEO_APPS)
  find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
  add_executable(digit_cnn_app inference_c/CNN/main.cpp)
  add_executable(digit_mlp_app inference_c/MLP/main.cpp)
  target_link_libraries(digit_cnn_app PRIVATE digit_cnn ${OpenCV_LIBS})
  target_link_libraries(digit_mlp_app PRIVATE digit_mlp ${OpenCV_LIBS})
  target_include_directories(digit_cnn_app PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_include_directories(digit_mlp_app PRIVATE ${OpenCV_INCLUDE_DIRS})
endif()
